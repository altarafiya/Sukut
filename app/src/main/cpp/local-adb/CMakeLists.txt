cmake_minimum_required(VERSION 3.22)
project(sukut_pairing LANGUAGES C CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../../../vendor/boringssl boringssl EXCLUDE_FROM_ALL)
add_library(sukut_pairing SHARED pairing.cpp)
target_link_libraries(sukut_pairing PRIVATE crypto)
target_compile_options(sukut_pairing PRIVATE -Os -fvisibility=hidden)
target_link_options(sukut_pairing PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL -Wl,--strip-all)
